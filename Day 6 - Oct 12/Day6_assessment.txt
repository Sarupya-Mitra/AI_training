CREATE OR REPLACE procedure get_total_spending_per_booking(
IN p_booking_id INT,
INOUT result_cursor REFCURSOR default 'total_spending')
LANGUAGE plpgsql
AS $$
BEGIN
	open result_cursor for
	Select distinct b.booking_id,g.guest_name, sum(bs.quantity * s.service_price) AS total_service_cost
	from guests g INNER JOIN bookings b
	ON g.guest_id = b.guest_id
	INNER JOIN booking_services bs on bs.booking_id = b.booking_id
	INNER JOIN services s on s.service_id = bs.service_id
	where b.booking_id = p_booking_id
	GROUP BY b.booking_id,g.guest_id;
END;
$$;

BEGIN;
CALL get_total_spending_per_booking(1002)
FETCH all from "total_spending"
COMMIT;
